import { exchangeCode } from "./github";
import { tokens } from "./token-store";

// GET /auth/callback?code=...&state=...
// Swaps the authorization code for an access token and keeps it in the session store.
export async function handleCallback(req, res) {
  const { code, state } = req.query;
  if (!code) return res.status(400).send("missing code");
  if (state !== req.session.oauthState) return res.status(400).send("state mismatch");

  const token = await exchangeCode(code);
  await tokens.save(req.session.id, token);
  delete req.session.oauthState;
  res.redirect("/");
}

import { createStore } from "../session";

// Access tokens live in the server-side session store, never in a cookie.
export const tokens = {
  store: createStore("oauth-tokens"),
  save(sessionId, token) {
    return this.store.set(sessionId, { token, savedAt: Date.now() });
  },
  get(sessionId) {
    return this.store.get(sessionId);
  },
};
